Key Findings

  • • W4A16 enables single-GPU inference (15.9 GB vs OOM for bf16)
  • • W4A16 is 3-5% faster than bf16 at same TP (less data to move)
  • • W4A16 TP=1 (8.23s) ≈ bf16 TP=4 (8.08s) — same latency, 4× fewer GPUs
  • • Memory savings: ~30% at TP=2, ~59% at TP=4
  • • No continuous batching — throughput flat regardless of concurrency
  • • Latency scales linearly with steps, quadratically with resolution

Quick Start (Python script)

# Show all configs without running
python benchmarks/diffusion/reproduce_flux1_dev.py --dry-run

# Run everything (bf16 + w4a16, all TP sizes)
python benchmarks/diffusion/reproduce_flux1_dev.py

# Run only W4A16 model
python benchmarks/diffusion/reproduce_flux1_dev.py --model w4a16

# Run only TP=4 configs for both models
python benchmarks/diffusion/reproduce_flux1_dev.py --tp 4

# Single config: bf16, TP=2, concurrency=1
python benchmarks/diffusion/reproduce_flux1_dev.py --model bf16 --tp 2 -c 1

Manual: bf16 TP=2

# Prerequisites
cd /home/yiliu7/workspace/vllm-omni
export LD_PRELOAD=/usr/lib/x86_64-linux-gnu/libnccl.so.2.29.7
PYTHON=/home/yiliu7/workspace/venvs/omni/bin/python
MODEL=/mnt/disk1/yiliu7/black-forest-labs/FLUX.1-dev

# Launch server
$PYTHON -m vllm_omni.entrypoints.cli.main serve $MODEL \
    --omni --tensor-parallel-size 2 --port 8099 --disable-log-stats &

# Wait for health
until curl -s http://localhost:8099/health; do sleep 5; done

# Benchmark
for C in 1 2 4 8; do
  $PYTHON benchmarks/diffusion/diffusion_benchmark_serving.py \
      --backend vllm-omni --base-url http://localhost:8099 \
      --model $MODEL --task t2i --dataset vbench \
      --num-prompts 10 --width 1024 --height 1024 \
      --num-inference-steps 20 --max-concurrency $C \
      --warmup-requests $C --warmup-concurrency $C \
      --output-file results/bf16_tp2_c${C}.json
done
